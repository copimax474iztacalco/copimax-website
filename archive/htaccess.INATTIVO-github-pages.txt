# Strategy: Clean URLs, Canonicalization, and Technical Consistency.

RewriteEngine On
RewriteBase /

# 1. FORCE HTTPS
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# 2. REDIRECT .html TO CLEAN URL (SEO Friendly)
# If the request is for an actual .html file, redirect to the clean version
RewriteCond %{THE_REQUEST} \s/+(.+)\.html[\s?] [NC]
RewriteRule ^ /%1 [R=301,L]

# 3. INTERNAL REWRITE: MAP CLEAN URL TO .html FILE
# If the file exists with .html extension, serve it
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_FILENAME}\.html -f
RewriteRule ^(.+)$ $1.html [L]
